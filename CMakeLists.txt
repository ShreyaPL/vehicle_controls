cmake_minimum_required(VERSION 3.22)
project(vehicle_controls VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(VC_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(VC_MUJOCO "Build native MuJoCo integration" ON)
option(VC_ROS2 "Build ROS 2 simulation adapter" OFF)
add_library(vc_options INTERFACE)
target_compile_options(vc_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(VC_SANITIZERS)
  target_compile_options(vc_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(vc_options INTERFACE -fsanitize=address,undefined)
endif()
add_library(vc_core src/control.cpp)
target_include_directories(vc_core PUBLIC include)
target_link_libraries(vc_core PUBLIC vc_options)
add_executable(truck_demo src/demo.cpp)
target_link_libraries(truck_demo PRIVATE vc_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE vc_core)
  foreach(case nominal limits stale invalid articulation latch prediction braking)
    add_test(NAME core_${case} COMMAND core_tests ${case})
  endforeach()
  add_test(NAME scenario_nominal COMMAND truck_demo)
  add_test(NAME scenario_dropout COMMAND truck_demo --dropout)
endif()
if(VC_MUJOCO)
  set(VC_PYTHON "${CMAKE_SOURCE_DIR}/.venv/bin/python" CACHE FILEPATH "Python with MuJoCo wheel")
  execute_process(COMMAND "${VC_PYTHON}" -c "import pathlib,mujoco; print(pathlib.Path(mujoco.__file__).parent)"
    OUTPUT_VARIABLE MUJOCO_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  find_path(MUJOCO_INCLUDE mujoco/mujoco.h PATHS "${MUJOCO_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(MUJOCO_LIBRARY NAMES mujoco libmujoco.so.3.3.7 PATHS "${MUJOCO_ROOT}" NO_DEFAULT_PATH REQUIRED)
  add_library(mujoco_native SHARED IMPORTED)
  set_target_properties(mujoco_native PROPERTIES IMPORTED_LOCATION "${MUJOCO_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MUJOCO_INCLUDE}")
  add_executable(mujoco_check src/mujoco_check.cpp)
  target_link_libraries(mujoco_check PRIVATE mujoco_native vc_options)
  if(BUILD_TESTING)
    add_test(NAME mujoco_native COMMAND mujoco_check "${CMAKE_SOURCE_DIR}/models/truck.xml")
  endif()
endif()
if(VC_ROS2)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  add_executable(truck_ros_sim src/ros_sim.cpp)
  target_link_libraries(truck_ros_sim PRIVATE vc_core)
  ament_target_dependencies(truck_ros_sim PUBLIC rclcpp geometry_msgs nav_msgs)
endif()
